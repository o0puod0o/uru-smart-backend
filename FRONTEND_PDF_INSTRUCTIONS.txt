STEP 2: CREATE PDF UTILITY HOOK
-------------------------------
Create file: src/hooks/useProfilePDF.js

```javascript
import html2pdf from 'html2pdf.js';
import RNFS from 'react-native-fs';
import Share from 'react-native-share';
import { API_URL } from '../config';

export const downloadProfilePDF = async (userId) => {
  try {
    // Fetch HTML from backend
    const response = await fetch(`${API_URL}/api/profile/${userId}/pdf`);
    const htmlContent = await response.text();

    // Convert HTML to PDF
    const element = document.createElement('div');
    element.innerHTML = htmlContent;

    const options = {
      margin: 10,
      filename: `profile-${userId}.pdf`,
      image: { type: 'png', quality: 0.98 },
      html2canvas: { scale: 2, logging: false },
      jsPDF: { orientation: 'portrait', unit: 'mm', format: 'a4' }
    };

    // Generate and save PDF
    await html2pdf()
      .setElement(element)
      .set(options)
      .save();

    return { success: true, message: 'PDF downloaded successfully' };

  } catch (error) {
    console.error('PDF Download Error:', error);
    return { success: false, error: error.message };
  }
};

export const shareProfilePDF = async (userId, userName) => {
  try {
    const response = await fetch(`${API_URL}/api/profile/${userId}/pdf`);
    const htmlContent = await response.text();

    // Convert to PDF first, then share
    const element = document.createElement('div');
    element.innerHTML = htmlContent;

    const pdfBlob = await html2pdf()
      .setElement(element)
      .outputPdf('blob');

    // Share using native share dialog
    await Share.open({
      title: 'Share Profile',
      message: `Check out ${userName}'s profile`,
      url: pdfBlob,
      type: 'application/pdf'
    });

  } catch (error) {
    console.error('Share Error:', error);
    throw error;
  }
};

export const previewProfileHTML = async (userId) => {
  try {
    const response = await fetch(`${API_URL}/api/profile/${userId}/pdf`);
    const htmlContent = await response.text();
    return htmlContent;
  } catch (error) {
    console.error('Preview Error:', error);
    return null;
  }
};
```


STEP 3: USE IN PROFILE SCREEN
-----------------------------
Create file: src/screens/ProfileScreen.js

```javascript
import React, { useState } from 'react';
import { View, Button, Alert, ActivityIndicator, ScrollView } from 'react-native';
import { WebView } from 'react-native-webview';
import { downloadProfilePDF, shareProfilePDF, previewProfileHTML } from '../hooks/useProfilePDF';

export default function ProfileScreen({ route }) {
  const { userId, userName } = route.params;
  const [loading, setLoading] = useState(false);
  const [htmlContent, setHtmlContent] = useState(null);
  const [showPreview, setShowPreview] = useState(false);

  const handleDownloadPDF = async () => {
    setLoading(true);
    try {
      const result = await downloadProfilePDF(userId);
      if (result.success) {
        Alert.alert('Success', result.message);
      } else {
        Alert.alert('Error', result.error);
      }
    } finally {
      setLoading(false);
    }
  };

  const handleSharePDF = async () => {
    setLoading(true);
    try {
      await shareProfilePDF(userId, userName);
      Alert.alert('Success', 'Profile shared!');
    } catch (error) {
      Alert.alert('Error', 'Failed to share profile');
    } finally {
      setLoading(false);
    }
  };

  const handlePreviewPDF = async () => {
    setLoading(true);
    try {
      const html = await previewProfileHTML(userId);
      setHtmlContent(html);
      setShowPreview(true);
    } finally {
      setLoading(false);
    }
  };

  if (showPreview && htmlContent) {
    return (
      <View style={{ flex: 1 }}>
        <WebView
          source={{ html: htmlContent }}
          style={{ flex: 1 }}
        />
        <View style={{ padding: 10, backgroundColor: '#f0f0f0' }}>
          <Button
            title="Close Preview"
            onPress={() => setShowPreview(false)}
          />
        </View>
      </View>
    );
  }

  return (
    <ScrollView style={{ flex: 1, padding: 20 }}>
      {/* Your existing profile content here */}

      {loading && <ActivityIndicator size="large" color="#0000ff" />}

      <View style={{ marginTop: 20, gap: 10 }}>
        <Button
          title="📥 Download Profile PDF"
          onPress={handleDownloadPDF}
          disabled={loading}
          color="#2d5016"
        />

        <Button
          title="👁️ Preview PDF"
          onPress={handlePreviewPDF}
          disabled={loading}
          color="#0066cc"
        />

        <Button
          title="📤 Share Profile"
          onPress={handleSharePDF}
          disabled={loading}
          color="#00aa00"
        />
      </View>
    </ScrollView>
  );
}
```

================================================================================
TROUBLESHOOTING
================================================================================

1. PDF not generating?
   - Check API_URL is correct
   - Make sure backend is running
   - Check browser console for errors

2. Thai characters not showing?
   - Backend HTML contains Thai text ✓ (already tested)
   - Issue is likely in html2pdf library
   - Alternative: Use native PDF library for iOS/Android

3. PDF too large?
   - Current size: 11KB HTML input
   - html2pdf adds encoding overhead
   - Normal size: 100-200KB PDF

4. Share not working?
   - Check react-native-share is installed
   - Need share permission in AndroidManifest.xml
