// hooks/useChatbot.js

import { useState, useCallback } from 'react';
import { API_URL } from '../config';

export const useChatbot = (token) => {
  const [messages, setMessages] = useState([]);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState(null);

  const sendMessage = useCallback(
    async (userMessage) => {
      try {
        setLoading(true);
        setError(null);

        // Add user message to UI
        setMessages((prev) => [
          ...prev,
          { role: 'user', content: userMessage },
        ]);

        // Build history for API
        const history = messages.map((msg) => ({
          role: msg.role,
          content: msg.content,
        }));

        // Send to backend
        const response = await fetch(`${API_URL}/api/chat`, {
          method: 'POST',
          headers: {
            'Authorization': `Bearer ${token}`,
            'Content-Type': 'application/json',
          },
          body: JSON.stringify({
            message: userMessage,
            history: history,
          }),
        });

        const data = await response.json();

        if (response.ok) {
          // Add bot reply to UI
          setMessages((prev) => [
            ...prev,
            { role: 'assistant', content: data.data.reply },
          ]);
        } else {
          setError(data.data.reply || 'เกิดข้อผิดพลาด');
          // Add error message to UI
          setMessages((prev) => [
            ...prev,
            { role: 'assistant', content: 'ขออภัย เกิดข้อผิดพลาด กรุณาลองใหม่' },
          ]);
        }
      } catch (err) {
        setError(err.message);
        setMessages((prev) => [
          ...prev,
          { role: 'assistant', content: 'เกิดข้อผิดพลาดในการเชื่อมต่อ' },
        ]);
      } finally {
        setLoading(false);
      }
    },
    [messages, token]
  );

  const clearMessages = useCallback(() => {
    setMessages([]);
    setError(null);
  }, []);

  return {
    messages,
    loading,
    error,
    sendMessage,
    clearMessages,
  };
};

// screens/ChatbotScreen.js

import React, { useState } from 'react';
import {
  View,
  Text,
  ScrollView,
  TextInput,
  TouchableOpacity,
  StyleSheet,
  ActivityIndicator,
  KeyboardAvoidingView,
} from 'react-native';
import { useChatbot } from '../hooks/useChatbot';

export default function ChatbotScreen({ token }) {
  const [inputText, setInputText] = useState('');
  const { messages, loading, error, sendMessage, clearMessages } =
    useChatbot(token);

  const handleSend = () => {
    if (inputText.trim()) {
      sendMessage(inputText.trim());
      setInputText('');
    }
  };

  return (
    <KeyboardAvoidingView style={styles.container} behavior="padding">
      {/* Header */}
      <View style={styles.header}>
        <Text style={styles.title}>🤖 ผู้ช่วยสนทนา</Text>
        <TouchableOpacity onPress={clearMessages}>
          <Text style={styles.clearBtn}>ล้างข้อความ</Text>
        </TouchableOpacity>
      </View>

      {/* Messages */}
      <ScrollView
        style={styles.messagesContainer}
        contentContainerStyle={styles.messagesList}
      >
        {messages.length === 0 ? (
          <View style={styles.emptyState}>
            <Text style={styles.emptyText}>
              สวัสดีครับ! ฉันเป็นผู้ช่วยสนทนาของมหาวิทยาลัย
            </Text>
            <Text style={styles.emptySubText}>
              ถามฉันเกี่ยวกับมหาวิทยาลัยราชภัฏอุตรดิตถ์
            </Text>
          </View>
        ) : (
          messages.map((msg, index) => (
            <View
              key={index}
              style={[
                styles.message,
                msg.role === 'user' ? styles.userMessage : styles.botMessage,
              ]}
            >
              <Text
                style={[
                  styles.messageText,
                  msg.role === 'user' ? styles.userText : styles.botText,
                ]}
              >
                {msg.content}
              </Text>
            </View>
          ))
        )}

        {loading && (
          <View style={styles.loadingContainer}>
            <ActivityIndicator size="large" color="#2d5016" />
            <Text style={styles.loadingText}>กำลังประมวลผล...</Text>
          </View>
        )}

        {error && (
          <View style={styles.errorContainer}>
            <Text style={styles.errorText}>⚠️ {error}</Text>
          </View>
        )}
      </ScrollView>

      {/* Input */}
      <View style={styles.inputContainer}>
        <TextInput
          style={styles.input}
          placeholder="พิมพ์คำถามของคุณ..."
          placeholderTextColor="#999"
          value={inputText}
          onChangeText={setInputText}
          editable={!loading}
          multiline
          maxLength={500}
        />
        <TouchableOpacity
          style={[styles.sendBtn, loading && styles.sendBtnDisabled]}
          onPress={handleSend}
          disabled={loading || !inputText.trim()}
        >
          <Text style={styles.sendBtnText}>ส่ง</Text>
        </TouchableOpacity>
      </View>
    </KeyboardAvoidingView>
  );
}

8. ERROR HANDLING
================================================================================

API Errors:
- 401 Unauthorized: Token expired or invalid
- 422 Unprocessable Entity: Invalid message format
- 500 Server Error: Gemini API issue
- Network Error: No connection to backend

Handle all with try-catch and show user-friendly messages.
